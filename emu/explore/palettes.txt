# Stage palettes (rom_map.yml pal_sets ... pal_cycles): every stage through the
# level select (FFF18B/FFF18A poked on the hero select), then Work RAM and CRAM.
# The palette target is FFE49A (4 lines x 16 words), the live buffer FFE51A, the
# cycle slots FFE5A0/FFE5AE/FFE5BC (byte 0 = cycle number | 80), the stage
# $C778/$C77A. Found: line 0 = 032410 and line 3 = pal_stage's palette for
# $C778/$C77A on every stage, but for the colours the slots' cycles write.
# Run: ga3 emu emu/explore/palettes.txt (about 13 x 1500 frames).
quiet

# stage 1
reset
include ../scenes/_boot.inc
press start 10
wait 90
poke8 FFF18B 01
poke8 FFF18A 00
press start 10
wait 30
poke8 FFF18B 01
poke8 FFF18A 00
wait 500
press start 10
wait 400
press b 6
wait 60
press b 6
wait 200
dump wram build/dumps/explore/palettes_1_wram.bin
dump cram build/dumps/explore/palettes_1_cram.bin
screenshot build/shots/explore/palettes_1.png 1
